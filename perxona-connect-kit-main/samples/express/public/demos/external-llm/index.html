Express + LLM skill

                

                  Request structured output on the server
                

                

                  Keep the API key and the trusted motion catalog in Express.
                  Select OpenAI-compatible Chat Completions or Claude Messages
                  API with LLM_PROVIDER. The LLM skill asks for
                  JSON, and server validation rejects invented motion IDs.
                

              

              
            

            
const LLM_PROVIDER = process.env.LLM_PROVIDER ?? "openai";
const LLM_BASE_URL = process.env.LLM_BASE_URL;
const LLM_API_KEY = process.env.LLM_API_KEY;
const LLM_MODEL = process.env.LLM_MODEL;

const scriptSchema = {
  name: "presenter_script",
  strict: true,
  schema: {
    type: "object",
    additionalProperties: false,
    required: ["reply", "script"],
    properties: {
      reply: { type: "string" },
      script: { type: "string" },
    },
  },
};

async function createDemoScript(prompt, motions) {
  const system = [
    "Write a short avatar script.",
    "Return JSON with reply and script only.",
    "Use Motion Markup [MOTION motion-id:1] only when useful.",
    "Never invent IDs; choose only from this catalog:",
    JSON.stringify(motions),
  ].join("\\n");
  const isClaude = LLM_PROVIDER === "anthropic";
  const response = await fetch(
    isClaude
      ? `${LLM_BASE_URL ?? "https://api.anthropic.com"}/v1/messages`
      : `${LLM_BASE_URL ?? "https://api.openai.com/v1"}/chat/completions`,
    {
    method: "POST",
    headers: {
      "Content-Type": "application/json",
      ...(isClaude
        ? {
            "anthropic-version": "2023-06-01",
            "x-api-key": LLM_API_KEY,
          }
        : { Authorization: `Bearer ${LLM_API_KEY}` }),
    },
    body: JSON.stringify(
      isClaude
        ? {
            model: LLM_MODEL,
            max_tokens: 1024,
            system: `${system}\\nUser request: ${prompt}`,
            messages: [{ role: "user", content: prompt }],
            output_config: {
              format: { type: "json_schema", schema: scriptSchema.schema },
            },
          }
        : {
            model: LLM_MODEL,
            response_format: { type: "json_schema", json_schema: scriptSchema },
            messages: [
              { role: "system", content: system },
              { role: "user", content: prompt },
            ],
          },
    ),
  },
  );

  const payload = await response.json();
  const content = isClaude
    ? payload.content.find(({ type }) => type === "text").text
    : payload.choices[0].message.content;
  const result = JSON.parse(content);
  return {
    reply: result.reply.trim(),
    script: validateDemoScript(result.script, motions),
  };
}

app.post("/api/demo-script", async (req, res) => {
  const motions = await getAvatarMotions(req.body.avatarId);
  res.json(await createDemoScript(req.body.prompt, motions));
});
